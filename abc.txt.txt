								modal.classList.add('active');
								// Trigger WooCommerce address field initialization

									jQuery(document.body).trigger('country_to_state_changed');

							}, 10);
						});

						modal.classList.remove('active');
						document.body.classList.remove('wc-modal-open');

						// If in edit mode (URL has params), redirect to clear them
						if (window.location.search.indexOf('edit_addr_idx') > -1) {
							var newUrl = window.location.href.split('?')[0];
							window.location.href = newUrl; // Reload to clear edit state
						}
					}

					if (closeBtn) closeBtn.addEventListener('click', closeModal);
					if (cancelBtn) cancelBtn.addEventListener('click', closeModal);

					// Close on outside click
					window.addEventListener('click', function (e) {
						if (e.target == modal) {
							closeModal();
						}
					});
				});
